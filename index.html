SELECT b.building_name,
  ROUND(AVG(u.utilization_pct),2) AS avg_util,
  COUNT(u.room_id) AS rooms
FROM buildings b
JOIN utilization u ON b.id = u.building_id
GROUP BY b.building_name
ORDER BY avg_util DESC;
